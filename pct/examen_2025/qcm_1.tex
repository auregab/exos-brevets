\begin{frame}{Question 1}

\begin{block}{}
Aux fins de l’attribution d’une date de dépôt international, il n’est pas exigé que :

\begin{itemize}
  \item[$\bullet$] a. la requête soit dûment remplie. 
  \item[$\bullet$] b. la signature d’au moins un déposant soit fournie. 
  \item[$\bullet$] c. les taxes de dépôt soient payées. 
  \item[$\circ$] d. Aucune des réponses proposées ci-dessus n’est correcte. 
\end{itemize}

\end{block}

\end{frame}

%--------------------------
\begin{frame}{Question 1}

\textbf{Base juridique :} \pctart{11}(1).  

\begin{itemize}
  \item[$\bullet$] \textbf{a.} \; Correct.

  \item[$\bullet$] \textbf{b.} \; Correct.

  \item[$\bullet$] \textbf{c.} \; Correct.

  \item[$\circ$] \textbf{d.} \; Faux puisque \textbf{a}, \textbf{b} et \textbf{c} sont correctes.
\end{itemize}

\end{frame}
